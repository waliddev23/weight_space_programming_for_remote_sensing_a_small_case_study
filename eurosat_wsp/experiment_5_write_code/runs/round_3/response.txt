Round 2 reached 49.0%, so this revision makes small, targeted adjustments rather than structural changes; as before, it has not been run.

Forest vs SeaLake: the split moved from 93/25 to 62/76, so I set the brightness and dark-block centroids about a third of the way back towards round 1 and gave Forest a small bias.
Highway: its centroid was sitting in low-texture residential territory (21 Residential images landed there, only 10 real highways). I moved it darker, bluer and less textured, where the true highways were actually going.
Dark mixed scenes going to Forest: I tightened the coarse-edge spread, since rivers and highways through forest have large-scale structure that pure forest lacks.
Biases: River’s penalty was too strong (recall fell to 35%), so I eased it. I reduced the Herbaceous boost and added small adjustments for Pasture, Permanent Crop and Residential.

import numpy as np

def build_weights():
    mean = np.array([0.485, 0.456, 0.406]); std = np.array([0.229, 0.224, 0.225])
    lw = std / 3.0                                   # luminance (pixel units) from normalised input
    c1w = np.zeros((16, 3, 3, 3)); c1b = np.zeros(16)
    for c in range(3):                               # ch0-2: raw R, G, B in [0,1]
        c1w[c, c, 1, 1] = std[c]; c1b[c] = mean[c]
    sx = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]]) / 8.0
    d1 = np.array([[0, 1, 2], [-1, 0, 1], [-2, -1, 0]]) / 8.0
    kernels = [sx, sx.T, d1, d1[::-1]]
    for i, k in enumerate(kernels):                  # ch3-10: |gradient| as +/- pairs
        for s, sign in enumerate((1.0, -1.0)):
            c1w[3 + 2 * i + s] = sign * lw[:, None, None] * k[None]
    c1w[11, :, 1, 1] = lw; c1b[11] = mean.mean() - 0.6   # ch11: bright pixels, relu(L-0.6)

    c2w = np.zeros((32, 16, 3, 3)); c2b = np.zeros(32)
    c2w[0, 0:3, 1, 1] = 1 / 3.0                                   # 0: L
    c2w[1, 2, 1, 1] = 1; c2w[1, 0, 1, 1] = -1; c2b[1] = 0.3       # 1: B-R (+0.3)
    c2w[2, 1, 1, 1] = 1; c2w[2, [0, 2], 1, 1] = -0.5; c2b[2] = 0.2  # 2: G-(R+B)/2 (+0.2)
    c2w[3, 3:11, 1, 1] = 1                                        # 3: texture energy T
    c2w[4, 3:11, 1, 1] = -1; c2b[4] = 0.04                        # 4: smooth blocks
    c2w[5, 3:11, 1, 1] = 1; c2b[5] = -0.09                        # 5: very textured blocks
    for i, k in enumerate((sx, -sx, sx.T, -sx.T)):                # 6-9: coarse edges of L
        c2w[6 + i, 0:3] = k[None] / 3.0
    c2w[10, 11, 1, 1] = 1                                         # 10: bright fraction
    c2w[11, 0:3, 1, 1] = -1 / 3.0; c2b[11] = 0.3                  # 11: dark blocks
    c2w[12, 3:11, 1, 1] = -1; c2b[12] = 0.02                      # 12: very smooth blocks (water)

    # derived features z = M f + o
    M = np.zeros((10, 32)); o = np.zeros(10)
    M[0, 0] = 1
    M[1, 1] = 1; o[1] = -0.3
    M[2, 2] = 1; o[2] = -0.2
    M[3, 3] = 1; M[4, 4] = 1; M[5, 5] = 1
    M[6, 6:10] = 1; M[6, 0] = -0.125                 # remove zero-padding border artefact
    M[7, 10] = 1; M[8, 11] = 1; M[9, 12] = 1

    # class centroids: AnnCrop Forest Herb Highway Industr Pasture PermCrop Resid River SeaLake
    mu = np.array([
        [.46, .223, .37, .34, .48, .32, .41, .40, .33, .24],       # L
        [-.02, .13, .03, .07, .04, .09, .01, .05, .10, .19],       # B-R
        [0, .02, .015, .01, -.015, .045, .005, -.01, .01, 0],      # green excess
        [.033, .035, .045, .048, .125, .028, .045, .087, .04, .022],  # T
        [.012, .016, .006, .006, 0, .016, .004, .001, .012, .027], # smooth
        [.003, 0, .002, .004, .045, 0, .002, .02, .002, 0],        # high texture
        [.03, .006, .018, .026, .045, .01, .02, .03, .03, .004],   # coarse edges
        [.015, 0, .004, .004, .03, 0, .006, .005, .003, 0],        # bright
        [.002, .08, .012, .02, .002, .02, .002, .005, .03, .07],   # dark
        [.001, .001, 0, 0, 0, .001, 0, 0, .001, .006],             # very smooth
    ]).T                                                           # [10, 10]
    s = np.array([.06, .04, .02, .025, .010, .025, .016, .02, .035, .004])
    prior = np.array([0, 0.3, 0.6, 0, -1.0, -0.4, 1.1, 0.3, -0.3, 0])  # over/under-prediction fix
    A = mu / s ** 2
    fcw = A @ M
    fcb = A @ o - 0.5 * (mu ** 2 / s ** 2).sum(1) + prior
    return {"conv1_weight": c1w, "conv1_bias": c1b,
            "conv2_weight": c2w, "conv2_bias": c2b,
            "fc_weight": fcw, "fc_bias": fcb}
